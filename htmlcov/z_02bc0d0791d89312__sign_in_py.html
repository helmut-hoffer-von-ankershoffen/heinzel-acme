1"""How the connectors say they are connected and signed in: a dummy sign-in (ADR 0136) and health checks (ADR 0055). 

2 

3A real company's gateway launcher holds the sign-in: `status` asks it who is signed in, `sign_in` starts its consent 

4flow and reports the consent page, `sign_out` asks it to forget the sign-in. Here the state is a setting, 

5`acme-connectors.sign_in_state` (`signed-in`, `signed-out` or `expired`), so every state can be shown and tested; 

6`sign_in` and `sign_out` change it for the running process only. Nothing reaches a network: the consent page is a 

7made-up address at acme.example. 

8"""

51class AcmeSignIn: 

52 """The gateway's sign-in, as a dummy whose state comes from `acme-connectors.sign_in_state`.""" 

53 

54 state: ClassVar[dict[str, str]] = {} 

55 """The state after a `sign_in` or `sign_out` in this process; the setting until then. Dummy only: a real plugin 

56 keeps no sign-in state of its own and asks its gateway launcher every time.""" 

57 

58 def __init__(self, settings: ConnectorSettings) -> None: 

59 """Read the starting state from the plugin's settings; the kernel passes them.""" 

60 self._settings = settings 

61 

62 @property 

63 def current(self) -> str: 

64 """`signed-in`, `signed-out` or `expired`.""" 

65 return self.state.get("now", self._settings.sign_in_state) 

66 

67 def status(self, *, live: bool) -> SignInStatus: 

68 """Who is signed in and which grants hold; `live` would ask the gateway once (the dummy asks nobody).""" 

69 held = {"signed-in": "granted", "signed-out": "missing", "expired": "unknown"}[self.current] 

70 grants = [GrantStatus(grant=grant, state=held, needed=grant.default) for grant in GRANTS] # type: ignore[arg-type] 

71 if self.current == "signed-out": 

72 return SignInStatus(signed_in=False, grants=grants, live=live) 

73 problem = "The ACME sign-in expired on 2026-11-13; sign in again." if self.current == "expired" else "" 

74 return SignInStatus(signed_in=True, account=ACCOUNT, grants=grants, problem=problem, live=live) 

75 

76 def sign_in( 

77 self, grants: frozenset[str], progress: Callable[[SignInProgress], None], stop: threading.Event 

78 ) -> SignInOutcome: 

79 """Report the gateway's consent page, then succeed at once unless `stop` is already set (a dummy).""" 

80 progress(SignInProgress(url=f"{CONSENT}?grants={'+'.join(sorted(grants))}", finish_on="anywhere")) 

81 if stop.is_set(): 

82 return SignInOutcome(state="cancelled", message="The ACME sign-in was cancelled; nothing changed.") 

83 self.state["now"] = "signed-in" 

84 return SignInOutcome(state="succeeded", message="Signed in to ACME's gateway.", account=ACCOUNT) 

85 

86 def sign_out(self) -> str: 

87 """Forget the sign-in, as the gateway launcher would; what happened.""" 

88 self.state["now"] = "signed-out" 

89 return "Signed out of ACME's gateway; reading through it stops until you sign in again."

92def checks(sign_in: AcmeSignIn) -> list[Check]: 

93 """Offline: every connector has its launcher, and the sign-in holds; nothing is called.""" 

94 launchers = [name for name, spec in CONNECTORS.items() if not spec.launcher] 

95 found: list[Check] = [ 

96 Check( 

97 name="launchers", 

98 status=Status.DOWN if launchers else Status.UP, 

99 detail=f"no launcher for {', '.join(launchers)}" if launchers else f"{len(CONNECTORS)} connectors", 

100 ) 

101 ] 

102 status = sign_in.status(live=False) 

103 login = Login(signed_in=status.signed_in, account=status.account) 

104 if not status.signed_in: 

105 found.append( 

106 Check( 

107 name="sign-in", 

108 status=Status.DOWN, 

109 detail="not signed in", 

110 fix="hz system connector connect acme-connectors", 

111 login=login, 

112 ) 

113 ) 

114 elif status.problem: 

115 found.append( 

116 Check( 

117 name="sign-in", 

118 status=Status.DEGRADED, 

119 detail=status.problem, 

120 fix="hz system connector connect acme-connectors", 

121 login=login, 

122 ) 

123 ) 

124 else: 

125 found.append(Check(name="sign-in", status=Status.UP, detail=f"signed in as {status.account}", login=login)) 

126 return found

129def live(sign_in: AcmeSignIn) -> list[Check]: 

130 """Live: start each connector, which checks that its server has every tool; here against the dummy servers.""" 

131 if sign_in.current != "signed-in": 

132 return [ 

133 Check( 

134 name="gateway", 

135 status=Status.DOWN, 

136 detail=f"not tested: {sign_in.current}", 

137 fix="hz system connector connect acme-connectors", 

138 ) 

139 ] 

140 found: list[Check] = [] 

141 for name in CONNECTORS: 

142 try: 

143 connect(name) 

144 except HeinzelError as exc: 

145 found.append(Check(name=f"connector {name}", status=Status.DOWN, detail=str(exc)[:200])) 

146 else: 

147 found.append(Check(name=f"connector {name}", status=Status.UP, detail="every tool there")) 

148 return found
